<script>
    const days = [0, 10, 20, 25, 30, 35, 40, 45];
    const brix = [2.0, 3.5, 7.0, 9.5, 12.0, 13.5, 14.2, 14.5]; // Cumulative sugar

    const ctx = document.getElementById('brixChart').getContext('2d');
    const brixChart = new Chart(ctx, {
        type: 'line',
        data: {
            labels: days,
            datasets: [{
                label: 'Sugar Level (Brix %)',
                data: brix,
                borderColor: '#10b981',
                backgroundColor: 'rgba(16, 185, 129, 0.1)',
                borderWidth: 3,
                pointRadius: 6,
                fill: true,
                tension: 0.2
            }]
        },
        options: {
            plugins: { legend: { display: false } },
            scales: {
                y: { title: { display: true, text: 'Sweetness (Brix %)' } },
                x: { title: { display: true, text: 'Days from Fruit Set' } }
            },
            onClick: (e) => {
                const activePoints = brixChart.getElementsAtEventForMode(e, 'nearest', { intersect: true }, true);
                if (activePoints.length) {
                    const idx = activePoints[0].index;
                    const d = days[idx];
                    const s = brix[idx];
                    
                    // Simple Derivative: (Current - Previous) / Time Gap
                    let rate = idx === 0 ? 0 : (brix[idx] - brix[idx-1]) / (days[idx] - days[idx-1]);
                    
                    let msg = "";
                    if (rate > 0.4) msg = "<strong>Peak Synthesis!</strong> The plant is working hard. The derivative is high!";
                    else if (d >= 40) msg = "<strong>Harvest Ready.</strong> The derivative is near zero; sugar gain has stopped.";
                    else msg = "<strong>Growth Phase.</strong> Developing size, not yet focusing on sugar.";

                    document.getElementById('output').innerHTML = `
                        <strong>Day ${d}:</strong> Sugar is ${s}%. <br>
                        <strong>Derivative f'(x):</strong> ${rate.toFixed(2)} Brix/Day (Speed). <br>
                        <hr style="border:0; border-top:1px solid #ccc; margin:10px 0;">
                        <em>Insight: ${msg}</em>
                    `;
                }
            }
        }
    });
</script>
